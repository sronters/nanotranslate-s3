cmake_minimum_required(VERSION 3.16)
project(nanomt_host LANGUAGES CXX)
add_executable(nanomt_cli main.cpp ../runtime/src/nanomt.cpp)
target_include_directories(nanomt_cli PRIVATE ../runtime/include)
target_compile_features(nanomt_cli PRIVATE cxx_std_17)
if(MSVC)
 target_compile_options(nanomt_cli PRIVATE /O2 /W4)
else()
 target_compile_options(nanomt_cli PRIVATE -O3 -Wall -Wextra -Wpedantic)
endif()
